Consultas Multitabla

Composición Interna (INNER JOIN)

SQL 
-- Alumnas matriculadas en Ingeniería Informática
SELECT DISTINCT p.nombre, p.apellido1, p.apellido2
FROM persona p
JOIN alumno_se_matricula_asignatura ama ON p.id = ama.id_alumno
JOIN asignatura a ON ama.id_asignatura = a.id
JOIN grado g ON a.id_grado = g.id
WHERE p.sexo = 'M' 
AND g.nombre = 'Grado en Ingeniería Informática (Plan 2015)';

-- Profesores con sus departamentos
SELECT p.apellido1, p.apellido2, p.nombre, d.nombre AS departamento
FROM persona p
JOIN profesor pr ON p.id = pr.id_profesor
JOIN departamento d ON pr.id_departamento = d.id
ORDER BY p.apellido1, p.apellido2, p.nombre;

Composición Externa (LEFT/RIGHT JOIN)

SQL 
-- Todos los profesores (incluyendo sin departamento)
SELECT d.nombre AS departamento, p.apellido1, p.apellido2, p.nombre
FROM persona p
JOIN profesor pr ON p.id = pr.id_profesor
LEFT JOIN departamento d ON pr.id_departamento = d.id
ORDER BY d.nombre, p.apellido1, p.apellido2, p.nombre;

-- Departamentos sin profesores
SELECT d.nombre
FROM departamento d
LEFT JOIN profesor p ON d.id = p.id_departamento
WHERE p.id_profesor IS NULL;

Consultas Resumen

Agregaciones Básicas

SQL 
-- Número total de alumnas
SELECT COUNT(*) AS total_alumnas
FROM persona
WHERE tipo = 'alumno' AND sexo = 'M';

-- Alumnos nacidos en 1999
SELECT COUNT(*) AS alumnos_1999
FROM persona
WHERE tipo = 'alumno' AND YEAR(fecha_nacimiento) = 1999;

Agrupamientos

SQL 
-- Profesores por departamento (solo departamentos con profesores)
SELECT d.nombre AS departamento, COUNT(p.id_profesor) AS num_profesores
FROM departamento d
JOIN profesor p ON d.id = p.id_departamento
GROUP BY d.nombre
ORDER BY num_profesores DESC;

-- Asignaturas por grado (incluyendo grados sin asignaturas)
SELECT g.nombre AS grado, COUNT(a.id) AS num_asignaturas
FROM grado g
LEFT JOIN asignatura a ON g.id = a.id_grado
GROUP BY g.nombre
ORDER BY num_asignaturas DESC;

Filtros sobre Agregaciones

SQL 
-- Grados con más de 40 asignaturas
SELECT g.nombre AS grado, COUNT(a.id) AS num_asignaturas
FROM grado g
JOIN asignatura a ON g.id = a.id_grado
GROUP BY g.nombre
HAVING COUNT(a.id) > 40
ORDER BY num_asignaturas DESC;

-- Alumnos matriculados por curso escolar
SELECT ce.anyo_inicio, COUNT(DISTINCT ama.id_alumno) AS alumnos_matriculados
FROM curso_escolar ce
LEFT JOIN alumno_se_matricula_asignatura ama ON ce.id = ama.id_curso_escolar
GROUP BY ce.anyo_inicio
ORDER BY ce.anyo_inicio;

Subconsultas

Subconsultas Escalares

SQL 
-- Alumno más joven
SELECT *
FROM persona
WHERE tipo = 'alumno' AND fecha_nacimiento = (
    SELECT MAX(fecha_nacimiento)
    FROM persona
    WHERE tipo = 'alumno'
);

-- Asignaturas con más créditos que la media
SELECT nombre, creditos
FROM asignatura
WHERE creditos > (
    SELECT AVG(creditos)
    FROM asignatura
);

Subconsultas con IN/NOT IN

SQL 
-- Profesores que imparten asignaturas
SELECT p.nombre, p.apellido1, p.apellido2
FROM persona p
WHERE p.id IN (
    SELECT id_profesor
    FROM asignatura
    WHERE id_profesor IS NOT NULL
);

-- Asignaturas nunca impartidas
SELECT nombre
FROM asignatura
WHERE id NOT IN (
    SELECT DISTINCT id_asignatura
    FROM alumno_se_matricula_asignatura
);

Subconsultas con EXISTS/NOT EXISTS

SQL 
-- Departamentos con profesores que imparten asignaturas
SELECT d.nombre
FROM departamento d
WHERE EXISTS (
    SELECT 1
    FROM profesor p
    JOIN asignatura a ON p.id_profesor = a.id_profesor
    WHERE p.id_departamento = d.id
);

-- Profesores sin asignaturas asignadas
SELECT p.nombre, p.apellido1, p.apellido2
FROM persona p
WHERE p.tipo = 'profesor' AND NOT EXISTS (
    SELECT 1
    FROM asignatura a
    WHERE a.id_profesor = p.id
);

Subconsultas en FROM

SQL 
-- Profesores y número de asignaturas que imparten
SELECT p.id, p.nombre, p.apellido1, p.apellido2, 
       COALESCE(a.num_asignaturas, 0) AS num_asignaturas
FROM persona p
LEFT JOIN (
    SELECT id_profesor, COUNT(*) AS num_asignaturas
    FROM asignatura
    GROUP BY id_profesor
) a ON p.id = a.id_profesor
WHERE p.tipo = 'profesor'
ORDER BY num_asignaturas DESC;

-- Departamentos y asignaturas no impartidas
SELECT d.nombre AS departamento, na.asignatura
FROM departamento d
JOIN (
    SELECT a.id_profesor, a.nombre AS asignatura, p.id_departamento
    FROM asignatura a
    JOIN profesor p ON a.id_profesor = p.id_profesor
    WHERE a.id NOT IN (
        SELECT id_asignatura
        FROM alumno_se_matricula_asignatura
    )
) na ON d.id = na.id_departamento;
